Architecture

The REPR Design Pattern

Single responsibility per operation: cohesive units of Request, Endpoint, and Response. 

// 1. JSON Body DTO with automated validation rules and OpenAPI documentation
type CreateUserRequest struct {
    Name  string `json:"name" doc:"Full name of user" example:"Jane Doe" validate:"required,min=3"`
    Email string `json:"email" doc:"Valid corporate or personal email" example:"jane.doe@example.com" validate:"required,email"`
    Role  string `json:"role" doc:"User access role" example:"admin" default:"member"`
}

// 2. Route Param DTO with automatic path parameter binding (:id)
type GetUserRequest struct {
    ID string `param:"id" doc:"User unique identifier" example:"usr_1"`
}

// Optional: Custom business validation (runs automatically after struct tags)
// func (r *CreateUserRequest) Validate() error { ... }
// Style A: Struct-based REPR Endpoint (ideal for testability & dependency injection)
type GetUserEndpoint struct{}

func (e *GetUserEndpoint) Handle(ctx *gota.Context, req GetUserRequest) (UserResponse, error) {
    for _, u := range mockUsers {
        if u.ID == req.ID {
            return u, nil
        }
    }
    return UserResponse{}, gota.NotFound("User not found")
}

userEndpoint := &GetUserEndpoint{}
gota.Get(app, "/users/:id", userEndpoint.Handle).
    Summary("Get user by ID").
    Tags("Users")

// Style B: Function-based REPR Endpoint (with automated validation & 201 Created)
gota.Post(app, "/users", func(ctx *gota.Context, req CreateUserRequest) (UserResponse, error) {
    newUser := UserResponse{ID: "usr_3", Name: req.Name, Email: req.Email, Role: req.Role}
    mockUsers = append(mockUsers, newUser)
    return newUser, nil
}).
    Status(201).
    Summary("Register user").
    Tags("Users")
// Strongly typed response contracts from main.go
type UserResponse struct {
    ID    string `json:"id" example:"usr_1"`
    Name  string `json:"name" example:"Jane Doe"`
    Email string `json:"email" example:"jane.doe@example.com"`
    Role  string `json:"role" example:"admin"`
}

type HealthResponse struct {
    Status  string `json:"status" example:"ok"`
    Version string `json:"version" example:"1.0.0"`
}

// Special response types supported natively:
// - gota.Empty{}       -> Renders HTTP 204 No Content (or empty GET /users)
// - gota.StreamFunc    -> Chunked byte stream (application/octet-stream)
// - gota.SSEStream     -> Server-Sent Events live stream (text/event-stream)
// - Declarative headers: Location string `header:"Location" json:"-" doc:"..."`
// Initializes Gota with built-in OpenAPI 3.1, Scalar UI, and TypeScript routes
isProd := os.Getenv("ENV") == "production"
docsPassword := os.Getenv("DOCS_PASSWORD") // Fallback "admin123" in dev

app := gota.New(gota.Config{
    Title:            "Gota REPR API",
    Version:          "1.0.0",
    Description:      "Go web framework oriented around the REPR Pattern with native interactive Scalar UI.",
    TypeScriptURL:    tsURL,    // "/types.d.ts" in dev, "-" in prod
    TypeScriptSDKURL: tsSDKURL, // "/sdk.ts" in dev, "-" in prod
})

// Protect documentation and SDK routes with BasicAuth, rate limiting, and HTTPS:
app.UseDocs(gota.BasicAuth("admin", docsPassword, gota.BasicAuthOptions{
    Realm: "Gota API Documentation",
    RateLimit: &gota.RateLimiterOptions{
        Max:    10,          // Max 10 attempts
        Window: time.Minute, // per minute per client IP
    },
    AllowWeakCredentials: true,   // silences warning in local dev
    RequireTLS:           isProd, // enforces HTTPS in production
}))

// Or export SDK offline directly to disk: app.ExportTypeScriptSDK("frontend/src/api.ts")
// 1. Contracts auto-generated by Gota from main.go Go structs:
export interface CreateUserRequest {
  name: string;   // required, min=3
  email: string;  // required, email
  role?: string;  // default: member
}

export interface UserResponse {
  id?: string;
  name?: string;
  email?: string;
  role?: string;
}

export interface GotaEndpoints {
  "GET /users": { response: UserResponse[] };
  "GET /users/:id": { params: { id: string }; response: UserResponse };
  "POST /users": { body: CreateUserRequest; response: UserResponse };
}

// 2. Frontend client usage (curl -u admin:admin123 -o src/api.ts http://localhost:3000/sdk.ts):
import { createGotaClient } from "./api";

const api = createGotaClient(process.env.NEXT_PUBLIC_API_URL || "http://localhost:3000");

// Autocompletes "GET /users/:id" and strictly enforces required params { id }:
const user = await api.fetch("GET /users/:id", {
  params: { id: "usr_1" },
});
console.log(user.name); // 100% typed: inferred directly from Go struct UserResponse

// Type-safe POST with autocompleted CreateUserRequest body:
const created = await api.fetch("POST /users", {
  body: { name: "Jane Doe", email: "jane.doe@example.com", role: "admin" },
});
Capabilities

Built for Production

Essential web framework primitives without third-party baggage.

Zero External Dependencies

Radix router, multi-source request binder, recursive struct validator, and fail-fast startup tag checks built purely on standard library primitives.

Scalar UI & OpenAPI 3.1

Interactive API reference at /docs with SRI-pinned Scalar, CSP hardening, route protection via app.UseDocs (with anti-brute-force rate limiting and anti-caching), and OpenAPI 3.1 schema at /openapi.json.

TypeScript Definitions & SDK

Zero-dependency end-to-end type safety from Go to frontend. Serves declarations at /types.d.ts and standalone client SDK at /sdk.ts using native fetch, or exports offline directly via app.ExportTypeScriptSDK.

Multi-Algorithm JWT

Zero-dependency signing, verification, and middleware for HS256, RS256, ES256, and EdDSA with PEM parsers and dynamic KeyFunc rotation.

Production Middlewares

CORS with automatic preflight, token bucket RateLimiter, SecureHeaders (HSTS & CSP), IPWhitelist (with CIDR subnets), BasicAuth, BearerToken, Timeout, Gzip, UUID v4 RequestID, structured SlogLogger, and panic Recover.

Native Streaming & SSE

Typed chunked streaming via gota.StreamFunc and real-time Server-Sent Events via gota.SSEStream with channel synchronization.

Graceful Shutdown

Signal trapping for SIGINT and SIGTERM: app.ListenGraceful completes in-flight requests before shutting down.

Quickstart

Up and Running in 60 Seconds

A complete REST API with auto-generated documentation in a single file.

main.go 
package main

import (
	"log"
	"os"
	"time"

	"github.com/carlosxfelipe/gota"
)

// ----------------------------------------------------------------------
// Request and Response Contracts (REPR Pattern)
// ----------------------------------------------------------------------

type CreateUserRequest struct {
	Name  string `json:"name" doc:"Full name of user" example:"Jane Doe" validate:"required,min=3"`
	Email string `json:"email" doc:"Valid corporate or personal email" example:"jane.doe@example.com" validate:"required,email"`
	Role  string `json:"role" doc:"User access role" example:"admin" default:"member"`
}

type GetUserRequest struct {
	ID string `param:"id" doc:"User unique identifier" example:"usr_1"`
}

type UserResponse struct {
	ID    string `json:"id" example:"usr_1"`
	Name  string `json:"name" example:"Jane Doe"`
	Email string `json:"email" example:"jane.doe@example.com"`
	Role  string `json:"role" example:"admin"`
}

type HealthResponse struct {
	Status  string `json:"status" example:"ok"`
	Version string `json:"version" example:"1.0.0"`
}

var mockUsers = []UserResponse{
	{ID: "usr_1", Name: "Jane Doe", Email: "jane.doe@example.com", Role: "admin"},
	{ID: "usr_2", Name: "John Smith", Email: "john.smith@example.com", Role: "member"},
}

// ----------------------------------------------------------------------
// Struct-Based REPR Endpoint Example
// ----------------------------------------------------------------------

type GetUserEndpoint struct{}

func (e *GetUserEndpoint) Handle(ctx *gota.Context, req GetUserRequest) (UserResponse, error) {
	for _, u := range mockUsers {
		if u.ID == req.ID {
			return u, nil
		}
	}
	return UserResponse{}, gota.NotFound("User not found")
}

// ----------------------------------------------------------------------
// Main Application
// ----------------------------------------------------------------------

func main() {
	isProd := os.Getenv("ENV") == "production"

	// In production, disable runtime HTTP TypeScript endpoints (use offline export instead)
	tsURL := "/types.d.ts"
	tsSDKURL := "/sdk.ts"
	if isProd {
		tsURL = "-"
		tsSDKURL = "-"
	}

	// Initializes Gota with built-in OpenAPI 3.1, Scalar docs, and TypeScript SDK routes
	app := gota.New(gota.Config{
		Title:            "Gota REPR API",
		Version:          "1.0.0",
		Description:      "Go web framework oriented around the REPR Pattern with native interactive Scalar UI.",
		TypeScriptURL:    tsURL,
		TypeScriptSDKURL: tsSDKURL,
	})

	// Global Middlewares
	app.Use(gota.RequestID())
	app.Use(gota.Logger())
	app.Use(gota.Recover())
	app.Use(gota.CORS())

	// Protect documentation, schemas, and SDK routes (/docs, /openapi.json, /types.d.ts, /sdk.ts)
	// with HTTP Basic Auth and built-in Rate Limiting to prevent brute-force attacks:
	docsPassword := os.Getenv("DOCS_PASSWORD")
	if docsPassword == "" {
		if isProd {
			log.Fatal("DOCS_PASSWORD environment variable must be set in production")
		}
		docsPassword = "admin123" // Fallback password for local development
	}
	app.UseDocs(gota.BasicAuth("admin", docsPassword, gota.BasicAuthOptions{
		Realm: "Gota API Documentation",
		RateLimit: &gota.RateLimiterOptions{
			Max:    10,          // Max 10 attempts
			Window: time.Minute, // per minute per client IP
		},
		AllowWeakCredentials: true,   // silences the weak password warning during local development
		RequireTLS:           isProd, // enforces HTTPS for remote clients in production
	}))

	// 1. Healthcheck Endpoint (Empty Request)
	gota.Get(app, "/", func(ctx *gota.Context, _ gota.Empty) (HealthResponse, error) {
		return HealthResponse{Status: "ok", Version: "1.0.0"}, nil
	}).
		Summary("Healthcheck").
		Description("Returns system status and current version").
		Tags("System")

	// 2. List Users (Empty Request)
	gota.Get(app, "/users", func(ctx *gota.Context, _ gota.Empty) ([]UserResponse, error) {
		return mockUsers, nil
	}).
		Summary("List all users").
		Tags("Users")

	// 3. Get User by ID (Struct-based REPR Endpoint)
	userEndpoint := &GetUserEndpoint{}
	gota.Get(app, "/users/:id", userEndpoint.Handle).
		Summary("Get user by ID").
		Tags("Users")

	// 4. Create User (Function-based REPR with automatic binding and validation)
	gota.Post(app, "/users", func(ctx *gota.Context, req CreateUserRequest) (UserResponse, error) {
		newUser := UserResponse{
			ID:    "usr_3",
			Name:  req.Name,
			Email: req.Email,
			Role:  req.Role,
		}
		mockUsers = append(mockUsers, newUser)
		return newUser, nil
	}).
		Status(201).
		Summary("Register user").
		Description("Validates input data automatically and returns the created user (201 Created)").
		Tags("Users")

	// 5. Classic route (without REPR structs, for quick hooks)
	app.Get("/ping", func(ctx *gota.Context) error {
		return ctx.Json(gota.Map{"message": "pong"})
	}).Summary("Ping Pong").Tags("System")

	if err := app.ListenGraceful(":3000"); err != nil {
		log.Printf("Server error: %v", err)
	}
}
