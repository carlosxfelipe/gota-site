Architecture

The REPR Design Pattern

Single responsibility per operation: cohesive units of Request, Endpoint, and Response. 

// Automated binding across route params, query, headers, form, and JSON body
type CreateUserRequest struct {
    Name  string `json:"name" doc:"Full Name" example:"Jane Doe" validate:"required,min=3" msg:"min:Name must have 3+ chars"`
    Email string `json:"email" doc:"Corporate email" example:"jane@corp.com" validate:"required,email"`
    Plan  string `query:"plan" doc:"Subscription tier" default:"Free" validate:"oneof=Free,Pro"`
}

// Optional: Custom business validation (also supports Validate(ctx *gota.Context) error)
func (r *CreateUserRequest) Validate() error {
    if strings.HasSuffix(r.Email, "@tempmail.com") {
        return errors.New("disposable emails are not permitted")
    }
    return nil
}
// Function-based generic endpoint with fluent Kite-like metadata
gota.Post(app, "/api/users", func(ctx *gota.Context, req CreateUserRequest) (UserResponse, error) {
    // 1. Parameter binding has already executed automatically
    // 2. Declarative tags and Validate() have already passed

    user := UserResponse{
        Location: "/api/users/usr_100",
        ID:       "usr_100",
        Name:     req.Name,
        Email:    req.Email,
    }

    return user, nil // Serialized as JSON with 201 Created (+ Location header)
}).
    Summary("Register new user").
    Description("Creates account and returns user profile").
    Tags("Users").
    Status(http.StatusCreated)
// Strongly typed response contract (with optional declarative response headers)
type UserResponse struct {
    Location string `header:"Location" json:"-" doc:"URI of created user"`
    ID       string `json:"id" example:"usr_100"`
    Name     string `json:"name" example:"Jane Doe"`
    Email    string `json:"email" example:"jane@corp.com"`
}

// Special response types supported natively:
// - gota.Empty{}       -> Renders HTTP 204 No Content
// - gota.StreamFunc    -> Chunked byte stream (application/octet-stream)
// - gota.SSEStream     -> Server-Sent Events live stream (text/event-stream)
// Interactive Scalar documentation served out of the box at /docs
// Schema exposed at /openapi.json

app := gota.New(gota.Config{
    Title:        "My API",
    Version:      "1.0.0",
    DocsURL:      "/docs",            // Default: /docs
    OpenAPIURL:   "/openapi.json",    // Default: /openapi.json
    ScalarTheme:  gota.ThemeFastify,  // Default: Fastify Theme
    ScalarLayout: "classic",          // Default: classic layout (Swagger-style)
})

// Zero YAML or JSON annotations required.
// OpenAPI 3.1 is derived directly from Go struct types and tags!
Capabilities

Built for Production

Essential micro-framework primitives without third-party baggage.

Zero External Dependencies

Radix router, multi-source request binder, recursive struct validator, and fail-fast startup tag checks built purely on standard library primitives.

Scalar UI & OpenAPI 3.1

Interactive API reference rendered via SRI-pinned Scalar with classic layout and Fastify theme at /docs, and full JSON Schema constraints at /openapi.json.

Multi-Algorithm JWT

Zero-dependency signing, verification, and middleware for HS256, RS256, ES256, and EdDSA with PEM parsers and dynamic KeyFunc rotation.

Production Middlewares

CORS with automatic OPTIONS preflight, token bucket RateLimiter, BasicAuth, UUID v4 RequestID, structured SlogLogger, and panic Recover.

Native Streaming & SSE

Typed chunked streaming via gota.StreamFunc and real-time Server-Sent Events via gota.SSEStream with channel synchronization.

Graceful Shutdown

Signal trapping for SIGINT and SIGTERM: app.ListenGraceful completes in-flight requests before shutting down.

Quickstart

Up and Running in 60 Seconds

A complete REST API with auto-generated documentation in a single file.

main.go 
package main

import (
    "log"
    "github.com/carlosxfelipe/gota"
)

type HelloRequest struct {
    Name string `query:"name" default:"World"`
}

type HelloResponse struct {
    Message string `json:"message"`
}

func main() {
    app := gota.New(gota.Config{
        Title:   "Hello API",
        Version: "1.0.0",
    })

    gota.Get(app, "/hello", func(ctx *gota.Context, req HelloRequest) (HelloResponse, error) {
        return HelloResponse{Message: "Hello, " + req.Name + "!"}, nil
    }).
        Summary("Say Hello").
        Tags("Greeting")

    log.Fatal(app.ListenGraceful(":3000"))
}
