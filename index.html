Architecture

The REPR Design Pattern

Single responsibility per operation: cohesive units of Request, Endpoint, and Response. 

// 1. Route Param DTO with automatic path parameter binding (:id)
type GetTodoRequest struct {
    ID string `param:"id" doc:"Todo identifier" example:"todo_1" validate:"required" msg:"Todo ID is required"`
}

// 2. JSON Body DTO with automated validation rules, custom messages, and OpenAPI metadata
type CreateTodoRequest struct {
    Title    string `json:"title" doc:"Task title" example:"Review pull requests" validate:"required,min=3" msg:"required:Title is required|min:Title must be at least 3 characters"`
    Priority string `json:"priority" doc:"Task priority" example:"high" default:"normal" validate:"oneof=low,normal,high" msg:"Priority must be low, normal, or high"`
}
type GetTodoEndpoint struct {
    Store *TodoStore
}

func (e *GetTodoEndpoint) Handle(ctx *gota.Context, req GetTodoRequest) (TodoResponse, error) {
    e.Store.mu.RLock()
    defer e.Store.mu.RUnlock()

    for _, t := range e.Store.items {
        if t.ID == req.ID {
            return t, nil
        }
    }
    return TodoResponse{}, gota.NotFound("Todo not found")
}

// Strongly typed response contract from main.go
type TodoResponse struct {
    ID       string `json:"id" doc:"Todo ID" example:"todo_1"`
    Title    string `json:"title" doc:"Task title" example:"Review pull requests"`
    Priority string `json:"priority" doc:"Task priority" example:"normal" validate:"oneof=low,normal,high"`
    Done     bool   `json:"done" doc:"Completion status" example:"false"`
}

// Special response types supported natively:
// - gota.Empty{}       -> Renders HTTP 204 No Content (or empty request DTO for GET /todos)
// - gota.StreamFunc    -> Chunked byte stream (application/octet-stream)
// - gota.SSEStream     -> Server-Sent Events live stream (text/event-stream)
// - Declarative headers: Location string `header:"Location" json:"-" doc:"..."`
isProd := os.Getenv("ENV") == "production"

// Disable runtime TypeScript HTTP routes in production (use app.ExportTypeScriptSDK in CI/CD instead)
tsURL, tsSDKURL := "/types.d.ts", "/sdk.ts"
if isProd {
    tsURL, tsSDKURL = "-", "-"
}

app := gota.New(gota.Config{
    Title:                 "Gota REPR API",
    Version:               "1.0.0",
    Description:           "Minimal Request-Endpoint-Response (REPR) API with Scalar OpenAPI 3.1 and TypeScript SDK.",
    TypeScriptURL:         tsURL,
    TypeScriptSDKURL:      tsSDKURL,
    DisallowUnknownFields: true,
    DocsCSPEnforce:        true,
})

// Protect /docs, /openapi.json, /types.d.ts, and /sdk.ts with Basic Auth and rate limiting
app.UseDocs(gota.BasicAuth("admin", docsPassword, gota.BasicAuthOptions{
    Realm:                "Gota API Documentation",
    RateLimit:            &gota.RateLimiterOptions{Max: 10, Window: time.Minute},
    AllowWeakCredentials: !isProd,
    RequireTLS:           isProd,
}))
// 1. Contracts auto-generated by Gota from main.go Go structs:
export interface CreateTodoRequest {
  title: string;                           // required, min=3
  priority?: "low" | "normal" | "high";  // default: normal
}

export interface TodoResponse {
  id: string;
  title: string;
  priority: "low" | "normal" | "high";
  done: boolean;
}

export interface GotaEndpoints {
  "GET /todos": { response: TodoResponse[] };
  "GET /todos/:id": { params: { id: string }; response: TodoResponse };
  "POST /todos": { body: CreateTodoRequest; response: TodoResponse };
}

Capabilities

Built for Production

Essential web framework primitives without third-party baggage.

Zero External Dependencies

Radix router, multi-source request binder, recursive struct validator, and fail-fast startup tag checks built purely on standard library primitives.

Scalar UI & OpenAPI 3.1

Interactive API reference at /docs with SRI-pinned Scalar, CSP hardening, route protection via app.UseDocs (with anti-brute-force rate limiting and anti-caching), and OpenAPI 3.1 schema at /openapi.json.

TypeScript Definitions & SDK

Zero-dependency end-to-end type safety from Go to frontend. Serves declarations at /types.d.ts and standalone client SDK at /sdk.ts using native fetch, or exports offline directly via app.ExportTypeScriptSDK.

Multi-Algorithm JWT

Zero-dependency signing, verification, and middleware for HS256, RS256, ES256, and EdDSA with PEM parsers and dynamic KeyFunc rotation.

Production Middlewares

CORS with automatic preflight, token bucket RateLimiter, SecureHeaders (HSTS & CSP), IPWhitelist (with CIDR subnets), BasicAuth, BearerToken, Timeout, Gzip, UUID v4 RequestID, structured SlogLogger, and panic Recover.

Native Streaming & SSE

Typed chunked streaming via gota.StreamFunc and real-time Server-Sent Events via gota.SSEStream with channel synchronization.

Graceful Shutdown

Signal trapping for SIGINT and SIGTERM: app.ListenGraceful completes in-flight requests before shutting down.

Quickstart

Up and Running in 60 Seconds

A complete REST API with auto-generated documentation in a single file.

main.go 
package main

import (
	"fmt"
	"log"
	"net/http"
	"os"
	"sync"
	"time"

	"github.com/carlosxfelipe/gota"
)

type TodoStore struct {
	mu    sync.RWMutex
	items []TodoResponse
}

func NewTodoStore() *TodoStore {
	return &TodoStore{
		items: []TodoResponse{
			{ID: "todo_1", Title: "Update README badges", Priority: "low", Done: true},
			{ID: "todo_2", Title: "Review pull requests", Priority: "normal", Done: false},
			{ID: "todo_3", Title: "Deploy release v1.0", Priority: "high", Done: false},
		},
	}
}

type GetTodoRequest struct {
	ID string `param:"id" doc:"Todo identifier" example:"todo_1" validate:"required" msg:"Todo ID is required"`
}

type CreateTodoRequest struct {
	Title    string `json:"title" doc:"Task title" example:"Review pull requests" validate:"required,min=3" msg:"required:Title is required|min:Title must be at least 3 characters"`
	Priority string `json:"priority" doc:"Task priority" example:"high" default:"normal" validate:"oneof=low,normal,high" msg:"Priority must be low, normal, or high"`
}

type TodoResponse struct {
	ID       string `json:"id" doc:"Todo ID" example:"todo_1"`
	Title    string `json:"title" doc:"Task title" example:"Review pull requests"`
	Priority string `json:"priority" doc:"Task priority" example:"normal" validate:"oneof=low,normal,high"`
	Done     bool   `json:"done" doc:"Completion status" example:"false"`
}

type ListTodosEndpoint struct {
	Store *TodoStore
}

func (e *ListTodosEndpoint) Handle(ctx *gota.Context, _ gota.Empty) ([]TodoResponse, error) {
	e.Store.mu.RLock()
	defer e.Store.mu.RUnlock()

	out := make([]TodoResponse, len(e.Store.items))
	copy(out, e.Store.items)
	return out, nil
}

type GetTodoEndpoint struct {
	Store *TodoStore
}

func (e *GetTodoEndpoint) Handle(ctx *gota.Context, req GetTodoRequest) (TodoResponse, error) {
	e.Store.mu.RLock()
	defer e.Store.mu.RUnlock()

	for _, t := range e.Store.items {
		if t.ID == req.ID {
			return t, nil
		}
	}
	return TodoResponse{}, gota.NotFound("Todo not found")
}

type CreateTodoEndpoint struct {
	Store *TodoStore
}

func (e *CreateTodoEndpoint) Handle(ctx *gota.Context, req CreateTodoRequest) (TodoResponse, error) {
	priority := req.Priority
	if priority == "" {
		priority = "normal"
	}

	e.Store.mu.Lock()
	defer e.Store.mu.Unlock()

	todo := TodoResponse{
		ID:       fmt.Sprintf("todo_%d", len(e.Store.items)+1),
		Title:    req.Title,
		Priority: priority,
		Done:     false,
	}
	e.Store.items = append(e.Store.items, todo)
	return todo, nil
}

func main() {
	isProd := os.Getenv("ENV") == "production"

	// Disable runtime TypeScript HTTP routes in production (use app.ExportTypeScriptSDK in CI/CD instead)
	tsURL, tsSDKURL := "/types.d.ts", "/sdk.ts"
	if isProd {
		tsURL, tsSDKURL = "-", "-"
	}

	app := gota.New(gota.Config{
		Title:                 "Gota REPR API",
		Version:               "1.0.0",
		Description:           "Minimal Request-Endpoint-Response (REPR) API with Scalar OpenAPI 3.1 and TypeScript SDK.",
		TypeScriptURL:         tsURL,
		TypeScriptSDKURL:      tsSDKURL,
		DisallowUnknownFields: true,
		DocsCSPEnforce:        true,
	})

	app.Use(gota.RequestID())
	app.Use(gota.Logger())
	app.Use(gota.Recover())
	app.Use(gota.SecureHeaders())
	app.Use(gota.Gzip())
	app.Use(gota.CORS())

	// Protect /docs, /openapi.json, /types.d.ts, and /sdk.ts with Basic Auth and rate limiting
	docsPassword := os.Getenv("DOCS_PASSWORD")
	if docsPassword == "" {
		if isProd {
			log.Fatal("DOCS_PASSWORD environment variable must be set in production")
		}
		docsPassword = "admin123"
	}
	app.UseDocs(gota.BasicAuth("admin", docsPassword, gota.BasicAuthOptions{
		Realm:                "Gota API Documentation",
		RateLimit:            &gota.RateLimiterOptions{Max: 10, Window: time.Minute},
		AllowWeakCredentials: !isProd,
		RequireTLS:           isProd,
	}))

	store := NewTodoStore()

	// Redirect root to Scalar UI and hide this route from OpenAPI/SDK
	app.Get("/", func(ctx *gota.Context) error {
		return ctx.Redirect("/docs")
	}).Hidden()

	gota.HandleEndpoint(app, http.MethodGet, "/todos", &ListTodosEndpoint{Store: store}).
		Summary("List all todos").
		Description("Returns all todo items in the store.").
		Tags("Todos")

	gota.HandleEndpoint(app, http.MethodGet, "/todos/:id", &GetTodoEndpoint{Store: store}).
		Summary("Get todo by ID").
		Description("Retrieves a single todo item by its unique identifier.").
		Tags("Todos")

	gota.HandleEndpoint(app, http.MethodPost, "/todos", &CreateTodoEndpoint{Store: store}).
		Status(http.StatusCreated).
		Summary("Create a new todo").
		Description("Validates the input payload and creates a new todo item.").
		Tags("Todos")

	if err := app.ListenGraceful(":3000"); err != nil {
		log.Printf("Server error: %v", err)
	}
}
